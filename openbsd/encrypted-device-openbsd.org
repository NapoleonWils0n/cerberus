** list disks

list disknames

#+begin_src sh
sysctl hw.disknames
#+end_src

Detailed Hardware Model & Size Information

#+begin_src sh
dmesg | grep -E '^(sd|wd|nvme|cd)[0-9]'
#+end_src

list active softraid volumes

#+begin_src sh
doas bioctl softraid0
#+end_src

** use dd to create a 128mb disk image

#+BEGIN_SRC sh
dd if=/dev/zero of=disk.img bs=1M count=128
#+END_SRC

** attach vnd loopback device

list attached loopback devices

#+begin_src sh
doas vnconfig -l
#+end_src

attach disk.img to a vnd loopback device

#+BEGIN_SRC sh
doas vnconfig vnd0 disk.img
#+END_SRC

check the disk.img is attached

#+begin_src sh
doas vnconfig -l
#+end_src

** initialize MBR and RAID partition on vnd0

Initialize default MBR partition table:

#+BEGIN_SRC sh
doas fdisk -iy vnd0
#+END_SRC

Write a disklabel spanning all free space as a RAID partition

#+BEGIN_SRC sh
echo 'RAID *' | doas disklabel -wAT- vnd0
#+END_SRC

** bioctl

initialize softraid crypto volume

Passphrase authentication

#+BEGIN_SRC sh
doas bioctl -c C -l vnd0a softraid0
#+END_SRC

Note the softraid pseudo-disk device name created in the output (e.g., sd2).

Attach an existing volume later:

#+BEGIN_SRC sh
doas bioctl -c C -l vnd0a softraid0
#+END_SRC

** newfs

Initialize an OpenBSD disklabel on the softraid volume (~sd2~) spanning partition ~a~:

#+BEGIN_SRC sh
doas disklabel -wA sd2
#+END_SRC

check the partion exists

#+begin_src sh
ls -l /dev/rsd2a
#+end_src

Format partition ~a~ with FFS (Fast File System):

#+BEGIN_SRC sh
doas newfs /dev/rsd2a
#+END_SRC

** mount

#+BEGIN_SRC sh
mkdir -p ~/mnt
doas mount /dev/sd2a /home/djwilcox/mnt
#+END_SRC

change the permissions

#+begin_src sh
doas chown -R djwilcox:djwilcox /home/djwilcox/mnt
#+end_src

Verify mount status:

#+BEGIN_SRC sh
df -h ~/mnt
#+END_SRC

** unmount

Unmount the file system

#+begin_src sh
doas umount /home/djwilcox/mnt
#+end_src

detach the softraid crypto volume

#+begin_src sh
doas bioctl -d sd2
#+end_src

unconfigure the loopback device:

#+BEGIN_SRC sh
doas vnconfig -u vnd0
#+END_SRC

#+begin_src sh
#!/bin/sh

#===============================================================================
# ossuary
#===============================================================================


#===============================================================================
# script usage
#===============================================================================

usage () {
# if argument passed to function echo it
[ -z "${1}" ] || echo "! ${1}"
echo "\
$(basename "$0") -m mount -f container
$(basename "$0") -m umount -f container"
exit 1
}

#===============================================================================
# error messages
#===============================================================================

ROOT_ERR="$(basename "$0") must be run as root"
WRONG_ARGS_ERR='wrong number of arguments passed to script'
INVALID_OPT_ERR='Invalid option:'
REQ_ARG_ERR='requires an argument'


#===============================================================================
# check if script is run as root
#===============================================================================

[ "$(id -u)" -eq 0 ] || usage "${ROOT_ERR}"


#===============================================================================
# check number of aruments passed to script
#===============================================================================

[ $# -gt 0 ] || usage "${WRONG_ARGS_ERR}"

#===============================================================================
# getopts check options passed to script
#===============================================================================

while getopts ':m:f:h' opt
do
  case ${opt} in
     m) action="${OPTARG}";;
     f) container="${OPTARG}";;
     h) usage;;
     \?) usage "${INVALID_OPT_ERR} ${OPTARG}" 1>&2;;
     :) usage "${INVALID_OPT_ERR} ${OPTARG} ${REQ_ARG_ERR}" 1>&2;;
  esac
done
shift $((OPTIND-1))


#===============================================================================
# functions
#===============================================================================

# mount function
mount_container () {
    [ -f "${container}" ] || usage "Container file not found"

    # attach the container to a vnd loopback device
    vnconfig vnd0 "${container}"

    # Attach the volume
    bioctl -c C -l vnd0a softraid0

    # mount the device
    mount /dev/sd2a /home/djwilcox/mnt

}

# umount function
umount_container () {
    [ -n "${container}" ] || usage "Container path required (-f)"

    # unmount device
    umount /home/djwilcox/mnt

    # detach the softraid crypto volume
    bioctl -d sd2

    # unconfigure the loopback device
    vnconfig -u vnd0
}

#===============================================================================
# mount and umount case statement
#===============================================================================

case "${action}" in
    mount) mount_container;;
    umount) umount_container;;
    *) usage;;
esac
#+end_src
