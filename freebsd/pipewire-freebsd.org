#+STARTUP: content
* pipewire freebsd
** packages

You need the following packages install

#+begin_src sh
doas pkg install grim slurp
#+end_src

slurp is use by the window picker to select the screen

** zshenv

You need to set the following in your ~/.zshenv config

#+begin_src sh
export XDG_CURRENT_DESKTOP=wlroots
#+end_src

** start pipewire

#+begin_src sh
pipewire &
/usr/local/libexec/xdg-desktop-portal -r &
/usr/local/libexec/xdg-desktop-portal-wlr -r &
#+end_src

** stop pipewire

#+begin_src sh
killall -9 xdg-desktop-portal xdg-desktop-portal-wlr pipewire 2>/dev/null
#+end_src

** wlr.portal

The wlr.portal is used when XDG_CURRENT_DESKTOP=wlroots is set
and you start screen sharing

#+begin_example
/usr/local/share/xdg-desktop-portal/portals/wlr.portal
#+end_example

#+begin_src conf
[portal]
DBusName=org.freedesktop.impl.portal.desktop.wlr
Interfaces=org.freedesktop.impl.portal.Screenshot;org.freedesktop.impl.portal.ScreenCast;
UseIn=wlroots;sway;Wayfire;river;phosh;Hyprland;
#+end_src

** Makefile

Makefile to start and stop Pipewire and XDG Portals for screen sharing.

To start Pipewire and the XDG Desktop Portals run:

#+begin_src sh
make
#+end_src

To stop Pipewire and the XDG Desktop Portals run:

#+begin_src sh
make clean
#+end_src

Pipewire Makefile for Wayland wlroots

#+begin_src makefile
#===============================================================================
# FILE: Makefile
# Project: Start Pipewire and xdg desktop portals on wlroots
# for OBS Studio and Browser Screen sharing
#===============================================================================

#===============================================================================
# Required Packages:
# doas pkg install pipewire wireplumber xdg-desktop-portal xdg-desktop-portal-wlr 
#===============================================================================

#===============================================================================
# Usage:
# Start Pipewire = make
# Stop Pipewire = make clean
#===============================================================================

#===============================================================================
# PHONY
#===============================================================================

.PHONY: all clean pipwire_start


#===============================================================================
# commands
#===============================================================================

# pipewire
PIPEWIRE_NAME  	:= pipewire
PIPEWIRE_EXEC  	:= /usr/local/bin/pipewire > /dev/null 2>&1 &

# xdg-desktop-portal
PORTAL_EXEC	:= /usr/local/libexec/xdg-desktop-portal -r > /dev/null 2>&1 &

# xdg-desktop-portal-wlr
PORTAL_WLR_NAME	:= xdg-desktop-portal-wlr
PORTAL_WLR_EXEC	:= /usr/local/libexec/xdg-desktop-portal-wlr -r > /dev/null 2>&1 &

# kill pipewire and xdg-desktop-portal-wlr
KILLALL		:= killall -9 $(PIPEWIRE_NAME) $(PORTAL_WLR_NAME) 2>/dev/null

#===============================================================================
# default target
#===============================================================================

all: pipewire_start


#===============================================================================
# pipewire start
#===============================================================================

pipewire_start: 
	@echo 'Starting Pipewire and XDG Portals'
	@$(PIPEWIRE_EXEC)
	@$(PORTAL_EXEC)
	@$(PORTAL_WLR_EXEC)


#===============================================================================
# clean 
#===============================================================================

clean:
	@echo 'Stopping Pipewire and XDG Portals'
	@$(KILLALL)
#+end_src
