;; note: run the following command to install the package
;; M-x package-vc-install RET https://github.com/sachac/whisper.el RET

(use-package whisper
  ;; Leverage the built-in package-vc features to clone the exact fork branch
  :vc (:url "https://github.com/sachac/whisper.el"
       :branch "master")
  :ensure t
  :config
  (setq whisper-install-whispercpp nil)
  (setq whisper-model "small")
  (setq whisper-language "en")
  (setq whisper-return-cursor-to-start nil)
  ;; Explicitly tell ffmpeg to use the Open Sound System driver for FreeBSD
  (setq whisper--ffmpeg-input-format "oss")
  ;; Point to your primary system microphone device (usually /dev/dsp)
  (setq whisper--ffmpeg-input-device "/dev/dsp3")

  ;; set the recording directory
  (defvar my-whisper-dir (expand-file-name "~/video/whisper/"))
  
  (defun my-whisper-set-temp-filename ()
    "Generate a unique timestamped filename inside the custom archive folder."
    ;; Create the directory dynamically if it doesn't exist yet
    (unless (file-directory-p my-whisper-dir)
      (make-directory my-whisper-dir t))
    (setq whisper--temp-file (expand-file-name
                              (format-time-string "%Y-%m-%d-%H-%M-%S.wav")
                              my-whisper-dir)))

  ;; execute whisper-before-transcription-hook with with-eval-after-load
  (setq whisper-before-transcription-hook nil)

  ;; ffmpeg wth rnnoise, noise gate, compressor and limiter
  (defun whisper--record-command (output-file)
      "Produces FFmpeg command to run with RNNoise and audio enhancement filters."
      (unless (executable-find "ffmpeg")
        (error "Needs FFmpeg to record audio"))
      (unless (or whisper--ffmpeg-input-file whisper--ffmpeg-input-format)
        (error "Set a suitable value for whisper--ffmpeg-input-format"))
      (unless (or whisper--ffmpeg-input-file whisper--ffmpeg-input-device)
        (error "Set a suitable value for whisper--ffmpeg-input-device"))
  
      `("ffmpeg"
        "-hide_banner"
        ,@(unless whisper--ffmpeg-input-file
            (list "-f" whisper--ffmpeg-input-format))
        "-i" ,(or whisper--ffmpeg-input-file whisper--ffmpeg-input-device)
        ,@(when (and (not whisper--ffmpeg-input-file) whisper-recording-timeout)
            (list "-t" (number-to-string whisper-recording-timeout)))
        "-af" ,(concat "arnndn=m=" (expand-file-name "~/.local/share/rnnoise/sh.rnnn") 
                       ",agate=threshold=0.05:range=0.03:attack=25:release=150"
                       ",acompressor=threshold=0.125:ratio=10:attack=6:release=60"
                       ",volume=7.5dB"
                       ",alimiter=level_out=0.79:release=60")
        "-ar" "16000"
        "-y" ,output-file))

;;----------------------------------------------------------------------------------
;; handle whisper commands with whisper-server
;;----------------------------------------------------------------------------------

(defun my-whisper-handle-commands (text)
  (interactive (list (if (region-active-p)
                         (buffer-substring (region-beginning) (region-end))
                       (buffer-substring (line-beginning-position) (line-end-position)))))
  ;; Clean up leading spaces, quotes, or punctuation returned by the server
  (setq text (replace-regexp-in-string "^[[:space:]\"',.!?]+" "" text))
  (let (done entry func num-args)
    (while (and text (not (string= text "")) (not done))
      (setq entry
            (seq-find
             (lambda (row)
               (string-match (format "^\\(?:%s\\)\\>" (car row)) (downcase text)))
             my-whisper-commands))
      (if (null entry)
          (setq done t)
        (setq func (cdr entry))
        (setq num-args (and func (func-arity func)))
        (cond
         ((commandp func)
          (setq text (replace-match "" nil nil text))
          (call-interactively func))
         ((and (functionp func) (>= (cdr num-args) 1))
          (unless (match-string 1 text)
            (setq text (replace-match "" nil nil text)))
          (setq text (funcall func (or (match-string 1 text) text))))
         ((functionp func)
          (setq text (replace-match "" nil nil text))
          (funcall func))))
      (when (string-match "^[[:space:]\\.?,]+\\(?:and \\)?" text)
        (setq text (replace-match "" nil nil text))))
    (when (string= text "") (setq text nil))
    (or text "")))

;; open side window and list voice commands and actions

(defun commands-sidebar ()
  "Display available Whisper voice commands in a neat side window on the right."
  (interactive)
  (let ((buf (get-buffer-create "*Whisper Commands*")))
    (with-current-buffer buf
      (let ((inhibit-read-only t))
        (erase-buffer)
        (insert "╔════════════════════════════╦════════════════════════════╗\n")
        (insert "║ Phrase                     ║ Action                     ║\n")
        (insert "╠════════════════════════════╬════════════════════════════╣\n")
        
        ;; Sort commands so bookmarks appear at the very bottom
        (let ((sorted-commands
               (sort (copy-sequence my-whisper-commands)
                     (lambda (a b)
                       (let ((a-bm (assoc (car a) bookmark-alist))
                             (b-bm (assoc (car b) bookmark-alist)))
                         (and (not a-bm) b-bm))))))
          
          (dolist (cmd sorted-commands)
            (let* ((phrase (car cmd))
                   (action-fn (cdr cmd))
                   (action-str
                    (cond
                     ((symbolp action-fn) (symbol-name action-fn))
                     ((assoc phrase bookmark-alist) (format "bookmark-jump \"%s\"" phrase))
                     (t "function")))
                   (line (format "║ %-26s ║ %-26s ║\n" 
                                 (truncate-string-to-width phrase 26 nil nil t)
                                 (truncate-string-to-width action-str 26 nil nil t))))
              (insert line))))
        
        (insert "╚════════════════════════════╩════════════════════════════╝\n")
        
        ;; Ensure it opens at the top of the buffer
        (goto-char (point-min))
        
        (special-mode))
      
      (display-buffer-in-side-window
       buf '((side . right)
             (window-width . 0.45))))))

;;----------------------------------------------------------------------------------
;; fix common voice errors
;;----------------------------------------------------------------------------------

(defvar my-common-edits
  '(
    ("dotfiles" "dot files")
    ("changelog" "change log")
    )
  "Commonly-misrecognized words or words that need special capitalization.")

(defun my-find-next-fix-point ()
  (when (re-search-forward
         (format "\\_<\\(%s\\)\\_>"
                 (downcase
                  (string-join
                   (seq-mapcat
                    (lambda (o)
                      (if (listp o)
                          (if (string= (car o) "") (cdr o) o)
                        (list o)))
                    my-common-edits)
                   "\\|")))
         nil t)
    (goto-char (match-beginning 0))
    (seq-find
     (lambda (o)
       (if (listp o)
           (seq-find (lambda (s)
                       (save-match-data
                         (string-match
                          (format "\\_<%s\\>" (downcase s))
                          (downcase (match-string 0))))) o)
         (string= (downcase o) (downcase (match-string 0)))))
     my-common-edits)))

(defun my-fix-common-errors ()
  (interactive)
  (let (done entry correction)
    (while (and
            (not done)
            (setq entry (my-find-next-fix-point)))
      (setq correction (if (listp entry) (car entry) entry))
      (if (let ((case-fold-search nil))
            (looking-at (regexp-quote correction)))
          (forward-word 1)
        (if (called-interactively-p 'any)
            (let* ((c (read-char (format "%s (yn.): " correction))))
              (cond
               ((= c ?y) (replace-match correction t t))
               ((= c ?n) (goto-char (match-end 0)))
               ((= c ?.) (setq done t))))
          (replace-match correction t t))))))

;; display recording icons in the buffer

(defvar whisper--current-overlay nil
  "Overlay to show whisper status at point.")

(advice-add 'whisper--setup-mode-line :override
  (lambda (command phase)
    (when whisper-show-progress-in-mode-line
      (with-current-buffer (or whisper--point-buffer (current-buffer))
        (if (eq command :show)
            (let ((pos (point)))
              (when (and whisper--current-overlay
                         (overlay-buffer whisper--current-overlay))
                (delete-overlay whisper--current-overlay))
              (setq whisper--current-overlay (make-overlay pos pos))
              (let* ((ind (whisper--mode-line-indicator phase))
                     (ind-string (cond
                                   ((stringp ind) ind)
                                   ((eq (car-safe ind) :eval) (eval (nth 1 ind)))
                                   (t ""))))
                ;; Removed the leading space inside the concat
                (overlay-put whisper--current-overlay 'after-string
                             (concat ind-string " "))))
          (when whisper--current-overlay
            (delete-overlay whisper--current-overlay)
            (setq whisper--current-overlay nil)))))))
