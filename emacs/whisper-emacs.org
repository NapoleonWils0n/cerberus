#+begin_src sh
doas pkg install cmake

**** Linux install cmake and other packages

#+begin_src sh
sudo apt install git cmake build-essential libvulkan-dev glslang-dev glslang-tools
#+end_src

**** build
***** change into the git repo

#+begin_src sh
cd whisper.cpp
#+end_src

***** create a directory for spirv

#+begin_src sh
mkdir -p ./external/spirv/unified1/
#+end_src

***** fetch spirv.hpp

Freebsd

#+begin_src sh
fetch -o ./external/spirv/unified1/spirv.hpp https://raw.githubusercontent.com/KhronosGroup/SPIRV-Headers/main/include/spirv/unified1/spirv.hpp
#+end_src

Linux

#+begin_src sh
curl -o ./external/spirv/unified1/spirv.hpp https://raw.githubusercontent.com/KhronosGroup/SPIRV-Headers/main/include/spirv/unified1/spirv.hpp
#+end_src

***** remove the build directory if it exists

#+begin_src sh
rm -rf build
#+end_src

***** Freebsd build whisper-cpp with vulkan

Configure CMake to build static libraries

#+begin_src sh
cmake -B build \
  -DGGML_VULKAN=ON \
  -DBUILD_SHARED_LIBS=OFF \
  -DWHISPER_ALL_EXTRAS=ON \
  -DCMAKE_CXX_FLAGS="-I$(pwd)/external"
#+end_src

Build the binary (adjust -j for your CPU cores)

#+begin_src sh
cmake --build build -j$(sysctl -n hw.ncpu) --config Release
#+end_src

***** Linux build whisper-cpp with vulkan

Configure CMake to build static libraries

#+begin_src sh
cmake -B build \
      -DGGML_VULKAN=1 \
      -DWHISPER_BUILD_TESTS=OFF \
      -DBUILD_SHARED_LIBS=OFF \
      -DCMAKE_BUILD_TYPE=Release
#+end_src

Build the binary (adjust -j for your CPU cores)

#+begin_src sh
cmake --build build -j$(nprocs) --config Release
#+end_src

***** install to whisper-cli to ~/bin

#+begin_src sh
cp build/bin/whisper-cli ~/bin/
#+end_src

***** install to whisper-server to ~/bin

#+begin_src sh
cp build/bin/whisper-server ~/bin/
#+end_src

  :config

;; phony additional commands
;; ~/.config/emacs/lisp/phony-speech-commands.el
(require 'phony-speech-commands) 

;; ----------------------------------------------------------------------------------
;; whisper-cpp
;; ----------------------------------------------------------------------------------

;; clone the whisper repository
;; mkdir -p ~/git/emacs
;; cd ~/git/emacs
;; git clone https://github.com/sachac/whisper.el.git

(use-package whisper
  ;; Leverage the built-in package-vc features to clone the exact fork branch
  :load-path "~/git/emacs/whisper.el"
  :ensure t
  :config
  (setq whisper-install-whispercpp nil)
  (setq whisper-model "small")
  (setq whisper-language "en")
  (setq whisper-return-cursor-to-start nil)
  ;; Explicitly tell ffmpeg to use the Open Sound System driver for FreeBSD
  (setq whisper--ffmpeg-input-format "oss")
  ;; Point to your primary system microphone device (usually /dev/dsp)
  (setq whisper--ffmpeg-input-device "/dev/dsp3")

  ;; set the recording directory
  (defvar my-whisper-dir (expand-file-name "/tmp/whisper/"))

  (defun my-whisper-set-temp-filename ()
    "Generate a unique timestamped filename inside the custom archive folder."
    ;; Create the directory dynamically if it doesn't exist yet (nil parents=t)
    (unless (file-directory-p my-whisper-dir)
      (make-directory my-whisper-dir t))
    (setq whisper--temp-file (expand-file-name
                              (format-time-string "%Y-%m-%d-%H-%M-%S.wav")
                              my-whisper-dir)))

  ;; execute whisper-before-transcription-hook with with-eval-after-load
  (setq whisper-before-transcription-hook nil)

  ;; ffmpeg wth rnnoise, noise gate, compressor and limiter
  (defun whisper--record-command (output-file)
      "Produces FFmpeg command to run with RNNoise and audio enhancement filters."
      (unless (executable-find "ffmpeg")
        (error "Needs FFmpeg to record audio"))
      (unless (or whisper--ffmpeg-input-file whisper--ffmpeg-input-format)
        (error "Set a suitable value for whisper--ffmpeg-input-format"))
      (unless (or whisper--ffmpeg-input-file whisper--ffmpeg-input-device)
        (error "Set a suitable value for whisper--ffmpeg-input-device"))
  
      `("ffmpeg"
        "-hide_banner"
        ,@(unless whisper--ffmpeg-input-file
            (list "-f" whisper--ffmpeg-input-format))
        "-i" ,(or whisper--ffmpeg-input-file whisper--ffmpeg-input-device)
        ,@(when (and (not whisper--ffmpeg-input-file) whisper-recording-timeout)
            (list "-t" (number-to-string whisper-recording-timeout)))
        "-af" ,(concat "arnndn=m=" (expand-file-name "~/.local/share/rnnoise/sh.rnnn") 
                       ",agate=threshold=0.05:range=0.03:attack=25:release=150"
                       ",acompressor=threshold=0.125:ratio=10:attack=6:release=60"
                       ",volume=7.5dB"
                       ",alimiter=level_out=0.79:release=60")
        "-ar" "16000"
        "-y" ,output-file))

(defun whisper--find-whispercpp-server ()
    (expand-file-name "~/bin/whisper-server"))

(defun whisper--model-file (&rest _)
    (expand-file-name "~/.local/share/whisper-models/ggml-small.en.bin"))

(setq whisper-server-mode 'local)) ;; whisper use package end


;; ----------------------------------------------------------------------------------
;; set whisper output directory
;; ----------------------------------------------------------------------------------

(with-eval-after-load 'whisper
  (add-hook 'whisper-before-transcription-hook #'my-whisper-set-temp-filename))


;; ----------------------------------------------------------------------------------
;; run commands or insert text
;; ----------------------------------------------------------------------------------

(with-eval-after-load 'whisper
  (setq whisper-insert-text-at-point
        '(speech-input-phony-process
          insert)))


;; ----------------------------------------------------------------------------------
;; Bind C-z globally to whisper-run, overriding Evil mode's default binding
;; ----------------------------------------------------------------------------------

(with-eval-after-load 'evil
  (define-key evil-normal-state-map (kbd "C-z") 'whisper-run)
  (define-key evil-insert-state-map (kbd "C-z") 'whisper-run)
  (define-key evil-visual-state-map (kbd "C-z") 'whisper-run)
  (define-key evil-motion-state-map (kbd "C-z") 'whisper-run))


;; ----------------------------------------------------------------------------------
;; display recording icons in the buffer
;; ----------------------------------------------------------------------------------

(defvar whisper--current-overlay nil
  "Overlay to show whisper status at point.")

(advice-add 'whisper--setup-mode-line :override
  (lambda (command phase)
    (when whisper-show-progress-in-mode-line
      (with-current-buffer (or whisper--point-buffer (current-buffer))
        (if (eq command :show)
            (let ((pos (point)))
              (when (and whisper--current-overlay
                         (overlay-buffer whisper--current-overlay))
                (delete-overlay whisper--current-overlay))
              (setq whisper--current-overlay (make-overlay pos pos))
              (let* ((ind (whisper--mode-line-indicator phase))
                     (ind-string (cond
                                   ((stringp ind) ind)
                                   ((eq (car-safe ind) :eval) (eval (nth 1 ind)))
                                   (t ""))))
                ;; Removed the leading space inside the concat
                (overlay-put whisper--current-overlay 'after-string
                             (concat ind-string " "))))
          (when whisper--current-overlay
            (delete-overlay whisper--current-overlay)
            (setq whisper--current-overlay nil)))))))
