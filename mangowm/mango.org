** swayfx build

#+begin_src sh
git clone -b 0.5 https://github.com/wlrfx/scenefx.git
cd scenefx
meson setup build -Dprefix=/usr/local
ninja -C build
doas ninja -C build install
#+end_src

** mango build

#+begin_src sh
git clone https://github.com/mangowm/mango.git
cd mango
meson setup build -Dprefix=/usr/local
ninja -C build
doas ninja -C build install
#+end_src

** mango config

#+begin_src sh
mkdir -p ~/.config/mango/
#+end_src

#+begin_src sh
cp /usr/local/etc/mango/config.toml ~/.config/mango/
#+end_src

** mangobar config

Before using mangobar run the following commands as user:

#+begin_src sh
mkdir -p ~/.config/mangobar
#+end_src

#+begin_src sh
cp /usr/local/share/mangobar/config.jsonc ~/.config/mangobar/config.jsonc
cp /usr/local/share/mangobar/style.css ~/.config/mangobar/style.css
#+end_src

** mango-start

#+begin_src sh
#!/bin/sh

# start mango with dbus
exec dbus-run-session mango
#+end_src

** check config

You can check your configuration for errors without starting mangowm:

#+begin_src sh
mango -c ~/.config/mango/config.conf -p
#+end_src
